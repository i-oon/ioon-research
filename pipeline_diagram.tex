\begin{tikzpicture}[node distance=1.0cm and 2.0cm]

%% ── Data source banner ───────────────────────────────────────────────────────
\node[srcbox] (src)
  {\textbf{Data:}\enspace Stick Insect Simulation \enspace|\enspace
   3 morphologies: short\,/\,medium\,/\,long leg\quad
   (third-person view camera — full scene: agent $+$ environment)};

%% ── Raw frames ───────────────────────────────────────────────────────────────
\node[framebox, below=1.1cm of src, xshift=-4.8cm] (ft)
  {$\boldsymbol{s_t}$ \,/\, $\mathrm{frame}_t$\\[4pt]
   \scriptsize RGB\;$224{\times}224{\times}3$};

\node[framebox, below=1.0cm of ft] (ft1)
  {$\boldsymbol{s_{t+1}}$ \,/\, $\mathrm{frame}_{t+1}$\\[4pt]
   \scriptsize RGB\;$224{\times}224{\times}3$};

%% ── Shared encoder (frozen) ──────────────────────────────────────────────────
\node[encbox, right=1.4cm of ft, yshift=-0.5cm] (enc)
  {\textbf{Visual Encoder}\\[3pt]
   \scriptsize V-JEPA2 ViT\\[1pt]
   {\color{gray!60}\scriptsize\textit{weights frozen}}};

%% ── Embeddings ───────────────────────────────────────────────────────────────
\node[embbox, right=1.4cm of enc, yshift=+0.56cm] (et)
  {$e_t$\\[3pt]
   \scriptsize $\mathbb{R}^{768}$\;|\;feature $t$};

\node[embbox, right=1.4cm of enc, yshift=-0.56cm] (et1)
  {$e_{t+1}$\\[3pt]
   \scriptsize $\mathbb{R}^{768}$\;|\;feature $t{+}1$};

%% ── IDM ──────────────────────────────────────────────────────────────────────
\node[modelbox, right=1.6cm of et, yshift=-0.56cm] (idm)
  {\textbf{Inverse Dynamics}\\[-1pt]\textbf{Model (IDM)}\\[4pt]
   \scriptsize MLP:\;$\mathbb{R}^{1536} \to \mathbb{R}^{k}$\\[1pt]
   \scriptsize in:\;$[e_t;\,e_{t+1}]$\enspace concat};

%% ── Red annotation near IDM ──────────────────────────────────────────────────
\node[redannot, above=0.45cm of idm] (annotidm)
  {What transition happened\\between these two states?};

%% ── Latent action z_t (highlighted) ─────────────────────────────────────────
\node[ztbox, right=1.6cm of idm] (zt)
  {\textbf{Latent Action}\;$z_t$\\[4pt]
   \scriptsize $\mathbb{R}^{k}$,\quad $k = 32$};

%% ── Anti-shortcut note ───────────────────────────────────────────────────────
\node[grayannot, above=0.4cm of zt] (anti)
  {\textbf{Anti-shortcut:}\\VQ-VAE bottleneck\\\textit{or} DeepLabCut $\Delta$kpt};

%% ── FDM ──────────────────────────────────────────────────────────────────────
\node[modelbox, below=2.0cm of idm] (fdm)
  {\textbf{Forward Dynamics}\\[-1pt]\textbf{Model (FDM)}\\[4pt]
   \scriptsize MLP:\;$\mathbb{R}^{768+k} \to \mathbb{R}^{768}$\\[1pt]
   \scriptsize in:\;$[e_t;\,z_t]$\enspace concat};

%% ── Predicted next embedding ─────────────────────────────────────────────────
\node[predbox, right=1.6cm of fdm] (ehat)
  {$\hat{e}_{t+1}$\;(predicted)\\[4pt]
   \scriptsize $\mathbb{R}^{768}$};

%% ── Reconstruction loss ──────────────────────────────────────────────────────
\node[lossbox, below=1.0cm of ehat] (lrecon)
  {$\mathcal{L}_{\mathrm{recon}}
      = \bigl\|\hat{e}_{t+1} - e_{t+1}\bigr\|^2$\\[5pt]
   {\scriptsize\color{green!50!black}%
     $\checkmark$\;\textbf{self-supervised — no action labels required}}};

%% ── Arrows: frames → encoder → embeddings ───────────────────────────────────
\draw[arr] (ft)  -- (enc);
\draw[arr] (ft1) -- (enc);
\draw[arr] (enc) -- (et);
\draw[arr] (enc) -- (et1);

%% ── Arrows: embeddings → IDM ─────────────────────────────────────────────────
\draw[arr] (et)  -- (idm);
\draw[arr] (et1) -- (idm);

%% ── Main flow (orange): IDM → z_t → FDM → ê_{t+1} → L_recon ────────────────
\draw[mainarr] (idm)  -- (zt);
\draw[mainarr] (zt)   |- (fdm);
\draw[arr]     (et)   |- (fdm);      %% e_t also enters FDM as current state
\draw[mainarr] (fdm)  -- (ehat);
\draw[mainarr] (ehat) -- (lrecon);

%% ── e_{t+1} → loss (reconstruction target) ───────────────────────────────────
\draw[arr] (et1) -| (lrecon);

%% ── Phase 1 border ───────────────────────────────────────────────────────────
\begin{scope}[on background layer]
  \node[phaseborder, fill=cBg,
        fit=(src)(ft)(ft1)(enc)(et)(et1)(idm)(zt)(fdm)(ehat)(lrecon)(anti)(annotidm),
        label={[phaselabel]above:\textbf{Phase 1 \;—\; Pretraining (label-free)}}
  ] (p1) {};
\end{scope}

%% ── PCA Validation box ───────────────────────────────────────────────────────
\node[validbox, below=1.4cm of p1.south, inner sep=13pt, align=center] (pca)
{%
  \textbf{Phase 1 Validation — PCA of $z_t$}\\[7pt]
  Collect $z_t$ from\;
  \textbf{short leg} (train)\;$+$\;
  \textbf{long leg} (train)\;$+$\;
  \textbf{medium leg} (unseen)\quad|\quad
  Behaviours:\;walk\;/\;turn\;/\;stop\\[5pt]
  Plot 2-D PCA coloured by \textbf{\{behaviour\}} and by \textbf{\{morphology\}}\\[6pt]
  \textcolor{green!50!black}{\textbf{Pass:}\enspace
    $z_t$ from different morphologies clusters by \emph{behaviour}, not by body shape}\\[3pt]
  \textcolor{red!60!black}{\textbf{Fail:}\enspace
    clusters split by morphology\;$\Rightarrow$\;anti-shortcut needs strengthening}%
};

\draw[thkarr] (p1.south) -- (pca.north);
